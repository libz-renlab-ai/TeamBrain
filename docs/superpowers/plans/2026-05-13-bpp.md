# Best-Practice Push (BPP) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the central team-brain server + AI mining pipeline + receiver inbox + lead console for "any team member's best practices auto-push to others" feature, per `docs/superpowers/specs/2026-05-13-best-practice-push-design.md`.

**Architecture:** Extend existing `digital-twin/mock-server.ts` with 4 new endpoints (`/v1/bp-push`, `/v1/inbox`, `/v1/inbox/act`, `/v1/revoke`); reuse `m5/secret-scanner`, `calibrator-v2/wilson`, `realtime-stream` SSE, `landing/overlay` dashboard scaffold. Storage on existing `<collectorDir>` JSONL pattern. AI mining is the only truly new component, runs server-side with a behavior-pattern miner over uploaded transcripts.

**Tech Stack:** Node.js (existing TeamBrain stack), TypeScript, vitest, Next.js (landing/overlay), Wilson lower-bound calibration, existing PII redactor + secret scanner. No new runtime deps for Phase 1.

---

## Plan Strategy: Detailed Phase 1, Backlog Phase 2-6

This plan **fully details Phase 1** (Server Foundation, 6 tasks). Phases 2-6 are listed as a **backlog** with one-line scope per task — they will be expanded into their own bite-sized plans when Phase 1 lands and dogfood feedback comes in. This matches the project's FIXEDFLOW pattern (one issue → one PR → one focused plan).

Reason: Pre-writing ~27 fully-detailed TDD tasks before Phase 1 ships would (a) lock decisions before learning from real code, (b) inflate plan to 2000+ lines, (c) risk drift between plan and reality. Phase 1 itself produces a fully working end-to-end MVP (push a BP, receive it, list inbox) and is what's worth detailed planning today.

```typescript
// packages/digital-twin/src/bpp/store.ts
import { writeFileSync, readFileSync, appendFileSync, existsSync, mkdirSync, readdirSync } from 'node:fs';
import { join, resolve as resolvePath } from 'node:path';
import type { BestPractice, InboxItem, PushEvent, TeamMember } from './types.js';

const ID_RE = /^[a-zA-Z0-9._-]+$/;

function assertSafeId(id: string): void {
  if (!ID_RE.test(id) || id.includes('..')) {
    throw new Error(`invalid id: ${id}`);
  }
}

function ensureDir(p: string): void {
  if (!existsSync(p)) mkdirSync(p, { recursive: true });
}

export function writeBp(rootDir: string, bp: BestPractice): void {
  assertSafeId(bp.id);
  const dir = resolvePath(rootDir, '_bp');
  ensureDir(dir);
  const target = resolvePath(dir, `${bp.id}.json`);
  if (!target.startsWith(dir)) throw new Error('path traversal');
  writeFileSync(target, JSON.stringify(bp, null, 2), 'utf8');
}

export function readBp(rootDir: string, id: string): BestPractice | null {
  assertSafeId(id);
  const target = resolvePath(rootDir, '_bp', `${id}.json`);
  if (!existsSync(target)) return null;
  return JSON.parse(readFileSync(target, 'utf8')) as BestPractice;
}

export function appendInbox(rootDir: string, item: InboxItem): void {
  assertSafeId(item.receiver_id.replace(/[@.]/g, '_'));
  const date = item.delivered_at.slice(0, 10);
  const receiverSafe = item.receiver_id.replace(/[^a-zA-Z0-9._-]/g, '_');
  const dir = resolvePath(rootDir, '_inbox', receiverSafe, date);
  ensureDir(dir);
  const file = resolvePath(dir, 'items.jsonl');
  appendFileSync(file, JSON.stringify(item) + '\n', 'utf8');
}

export function listInbox(rootDir: string, receiverId: string): InboxItem[] {
  const receiverSafe = receiverId.replace(/[^a-zA-Z0-9._-]/g, '_');
  const dir = resolvePath(rootDir, '_inbox', receiverSafe);
  if (!existsSync(dir)) return [];
  const out: InboxItem[] = [];
  for (const dateDir of readdirSync(dir).sort()) {
    const file = resolvePath(dir, dateDir, 'items.jsonl');
    if (!existsSync(file)) continue;
    const lines = readFileSync(file, 'utf8').split('\n').filter(Boolean);
    for (const ln of lines) out.push(JSON.parse(ln) as InboxItem);
  }
  return out.sort((a, b) => a.delivered_at.localeCompare(b.delivered_at));
}

export function appendAudit(rootDir: string, ev: PushEvent): void {
  const date = ev.timestamp.slice(0, 10);
  const dir = resolvePath(rootDir, '_audit');
  ensureDir(dir);
  const file = resolvePath(dir, `${date}.jsonl`);
  appendFileSync(file, JSON.stringify(ev) + '\n', 'utf8');
}

export function writeMember(rootDir: string, m: TeamMember): void {
  assertSafeId(m.user_id.replace(/[@.]/g, '_'));
  const dir = resolvePath(rootDir, '_team');
  ensureDir(dir);
  const file = resolvePath(dir, 'members.json');
  const existing: TeamMember[] = existsSync(file) ? JSON.parse(readFileSync(file, 'utf8')) : [];
  const next = [...existing.filter((x) => x.user_id !== m.user_id), m];
  writeFileSync(file, JSON.stringify(next, null, 2), 'utf8');
}

export function readMembers(rootDir: string): TeamMember[] {
  const file = resolvePath(rootDir, '_team', 'members.json');
  if (!existsSync(file)) return [];
  return JSON.parse(readFileSync(file, 'utf8')) as TeamMember[];
}
```

```typescript
// packages/digital-twin/src/bpp/server-handlers.ts
import { randomUUID } from 'node:crypto';
import { writeBp, appendInbox, listInbox, appendAudit } from './store.js';
import type { BestPractice, InboxItem, PushEvent } from './types.js';

export interface BpPushBody {
  bp: BestPractice;
  receivers: string[];
}

export interface BpPushResult {
  ok: true;
  bp_id: string;
  delivered_to: string[];
}

export function handleBpPush(rootDir: string, body: BpPushBody): BpPushResult {
  if (!body || !body.bp || !body.bp.id || typeof body.bp.title !== 'string') {
    throw new Error('malformed body: bp + bp.id + bp.title required');
  }
  if (!Array.isArray(body.receivers)) {
    throw new Error('malformed body: receivers must be array');
  }
  writeBp(rootDir, body.bp);
  const now = new Date().toISOString();
  for (const rcvr of body.receivers) {
    const item: InboxItem = {
      schema_version: 1,
      id: `inbox-${rcvr.replace(/[^a-z0-9]/gi, '_')}-${randomUUID().slice(0, 8)}`,
      receiver_id: rcvr,
      bp_id: body.bp.id,
      status: 'pending',
      delivered_at: now,
      acted_at: null,
      forced_by_lead: false,
      delivery_channels: ['statusline', 'dashboard'],
    };
    appendInbox(rootDir, item);
  }
  const ev: PushEvent = {
    schema_version: 1,
    id: `ev-${randomUUID().slice(0, 8)}`,
    event_type: 'pushed',
    bp_id: body.bp.id,
    actor: 'system',
    timestamp: now,
    metadata: { receivers: body.receivers },
  };
  appendAudit(rootDir, ev);
  return { ok: true, bp_id: body.bp.id, delivered_to: body.receivers };
}

export interface InboxResult {
  ok: true;
  items: InboxItem[];
}

export function handleInbox(rootDir: string, receiverId: string): InboxResult {
  return { ok: true, items: listInbox(rootDir, receiverId) };
}
```

```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { mkdtempSync } from 'node:fs';
import { join } from 'node:path';

import { startMockServer, type MockServerHandle } from '../../mock-server.js';

  let dir: string;

  beforeAll(async () => {

  });

      example: 'example',
```
